const PLANNERS = {
  '10200366': { name:'반현기', branch:'대전 중구지사', phone:'01030413282', avatar:'반' }
};
const params = new URLSearchParams(window.location.search);
const pid = params.get('p') || '10200366';
const pl = PLANNERS[pid] || PLANNERS['10200366'];
document.getElementById('plannerName').textContent = pl.name + ' 플래너';
document.getElementById('plannerSub').textContent = '사번 ' + pid + ' · ' + pl.branch;
document.getElementById('avatarEl').textContent = pl.avatar;
document.getElementById('callBtn').addEventListener('click', function() {
  window.location.href = 'tel:' + pl.phone;
});
document.getElementById('smsBtn').addEventListener('click', function() {
  document.getElementById('smsModal').classList.add('open');
  setTimeout(function(){ document.getElementById('custName').focus(); }, 150);
});
document.getElementById('cancelBtn').addEventListener('click', function() {
  document.getElementById('smsModal').classList.remove('open');
});
document.getElementById('sendBtn').addEventListener('click', function() {
  const name = document.getElementById('custName').value.trim();
  const msg = document.getElementById('custMsg').value.trim();
  if (!name) { alert('이름을 입력해주세요'); return; }
  const body = '[청호나이스 상담신청]\n고객명: ' + name + '\n\n' + (msg || '상담 문의드립니다.');
  const isIOS = /iPad|iPhone|iPod/.test(navigator.userAgent);
  const sep = isIOS ? '&' : '?';
  window.location.href = 'sms:' + pl.phone + sep + 'body=' + encodeURIComponent(body);
});
